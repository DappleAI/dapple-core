% ============================================
% CUSTOM TITLE PAGE
% ============================================
% Each guide sets these commands before \maketitle:
%   \dappletitle{Title}
%   \dapplesubtitle{Subtitle}
%   \dapplesubsubtitle{For whom} (optional)
%   \dapplefooter{Footer text}

\makeatletter
\newcommand{\dappletitle}[1]{\def\@dappletitle{#1}}
\newcommand{\dapplesubtitle}[1]{\def\@dapplesubtitle{#1}}
\newcommand{\dapplesubsubtitle}[1]{\def\@dapplesubsubtitle{#1}}
\newcommand{\dapplefooter}[1]{\def\@dapplefooter{#1}}

% Defaults
\dappletitle{\@title}
\dapplesubtitle{}
\dapplesubsubtitle{}
\dapplefooter{dapple-prompt Documentation}
\makeatother

\makeatletter
\renewcommand{\maketitle}{%
  \begin{titlepage}
    \begin{tikzpicture}[remember picture,overlay]
      % Green header band
      \fill[dapplegreen] (current page.north west) rectangle ([yshift=-4cm]current page.north east);
      % Light green accent stripe
      \fill[dapplegreenlight] ([yshift=-4cm]current page.north west) rectangle ([yshift=-4.15cm]current page.north east);
    \end{tikzpicture}

    \vspace*{1cm}

    % Logo
    \begin{center}
      \includegraphics[width=2.5cm]{logo.pdf}
    \end{center}

    \vspace{2cm}

    % Title
    \begin{center}
      {\fontsize{36}{42}\selectfont\bfseries\sffamily\color{dapplegreen}\@dappletitle\par}
      \vspace{0.8cm}
      {\Large\color{dapplegrey}\@dapplesubtitle\par}
      \ifx\@dapplesubsubtitle\empty\else
        {\large\color{dapplegrey}\@dapplesubsubtitle\par}
      \fi
    \end{center}

    \vfill

    % Decorative line
    \begin{center}
      \begin{tikzpicture}
        \draw[dapplegreen, line width=2pt] (0,0) -- (8,0);
        \fill[dapplegreen] (4,0) circle (4pt);
      \end{tikzpicture}
    \end{center}

    \vspace{1cm}

    % Author and version info
    \begin{center}
      {\large\sffamily\color{dapplegrey}\@author\par}
      \vspace{0.5cm}
      {\normalsize\color{gray}\@date\par}
    \end{center}

    \vspace{2cm}

    % Footer band
    \begin{tikzpicture}[remember picture,overlay]
      \fill[dapplegreenxlight] (current page.south west) rectangle ([yshift=1.5cm]current page.south east);
      \node[anchor=south] at ([yshift=0.5cm]current page.south) {\color{dapplegrey}\small\sffamily dapple-prompt Documentation};
    \end{tikzpicture}
  \end{titlepage}
}
\makeatother

% ============================================
% CHAPTER AND SECTION STYLING
% ============================================
% Chapter: Green title with decorative underline
\newcommand{\chapmark}{}
\titleformat{\chapter}{\normalfont\Huge\bfseries\sffamily\color{dapplegreen}}{%
  \colorbox{dapplegreenxlight}{\makebox[2.5em]{\color{dapplegreen}\thechapter}}%
}{20pt}{\renewcommand{\chapmark}}
\titlespacing*{\chapter}{0pt}{0pt}{20pt}

% Add decorative rule after chapter title
\newcommand{\chapterrule}{%
  \par\nobreak\vspace{10pt}%
  \hbox to \textwidth{\color{dapplegreen}\leaders\hrule height 2pt\hfill}%
  \par\nobreak\vspace{20pt}%
}

% Section: Green with subtle underline
\titleformat{\section}{\normalfont\Large\bfseries\sffamily\color{dapplegreen}}{%
  \colorbox{dapplegreenlight}{\makebox[1.8em]{\color{white}\thesection}}%
}{0.8em}{}
\titlespacing*{\section}{0pt}{2em}{0.8em}

% Subsection: Grey with green number box
\titleformat{\subsection}{\normalfont\large\bfseries\sffamily\color{dapplegrey}}{%
  \fcolorbox{dapplegreen}{white}{\makebox[2.2em]{\color{dapplegreen}\thesubsection}}%
}{0.8em}{}
\titlespacing*{\subsection}{0pt}{1.5em}{0.5em}

% Subsubsection: Smaller, with green accent
\titleformat{\subsubsection}{\normalfont\normalsize\bfseries\sffamily\color{dapplegrey}}{%
  \textcolor{dapplegreen}{\thesubsubsection}%
}{0.8em}{}
